CREATE DATABASE IF NOT EXISTS ride_booking_db;
USE ride_booking_db;

CREATE TABLE riders (
    rider_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(15) UNIQUE,
    city VARCHAR(50)
);

CREATE TABLE drivers (
    driver_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(15) UNIQUE,
    license_number VARCHAR(50) UNIQUE NOT NULL,
    vehicle_type VARCHAR(50),
    vehicle_number VARCHAR(20) UNIQUE,
    city VARCHAR(50)
);

CREATE TABLE trips (
    trip_id INT AUTO_INCREMENT PRIMARY KEY,
    rider_id INT NOT NULL,
    driver_id INT NOT NULL,
    pickup_location VARCHAR(100) NOT NULL,
    drop_location VARCHAR(100) NOT NULL,
    trip_date DATETIME NOT NULL,
    distance_km DECIMAL(6,2) NOT NULL,
    FOREIGN KEY (rider_id) REFERENCES riders(rider_id),
    FOREIGN KEY (driver_id) REFERENCES drivers(driver_id)
);

CREATE TABLE fares (
    fare_id INT AUTO_INCREMENT PRIMARY KEY,
    trip_id INT NOT NULL,
    base_fare DECIMAL(10,2) NOT NULL,
    per_km_rate DECIMAL(10,2) NOT NULL,
    distance_charge DECIMAL(10,2) NOT NULL,
    total_fare DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (trip_id) REFERENCES trips(trip_id)
);

CREATE TABLE ratings (
    rating_id INT AUTO_INCREMENT PRIMARY KEY,
    trip_id INT NOT NULL,
    rider_id INT NOT NULL,
    driver_id INT NOT NULL,
    rating INT NOT NULL,
    review VARCHAR(255),
    FOREIGN KEY (trip_id) REFERENCES trips(trip_id),
    FOREIGN KEY (rider_id) REFERENCES riders(rider_id),
    FOREIGN KEY (driver_id) REFERENCES drivers(driver_id),
    CHECK (rating BETWEEN 1 AND 5)
);

-- REQUIRED SQL QUERIES
-- =========================

-- 1. JOIN
SELECT 
    t.trip_id,
    r.name AS rider_name,
    d.name AS driver_name,
    t.pickup_location,
    t.drop_location,
    t.distance_km
FROM trips t
JOIN riders r ON t.rider_id = r.rider_id
JOIN drivers d ON t.driver_id = d.driver_id;


-- 2. GROUP BY + HAVING
SELECT 
    pickup_location,
    COUNT(*) AS total_rides,
    AVG(distance_km) AS average_distance
FROM trips
GROUP BY pickup_location
HAVING COUNT(*) >= 2;


-- 3. SUBQUERY
SELECT 
    trip_id,
    rider_id,
    driver_id,
    distance_km
FROM trips
WHERE distance_km > (
    SELECT AVG(distance_km)
    FROM trips
);


-- 4. VIEW
CREATE VIEW ride_details AS
SELECT 
    t.trip_id,
    r.name AS rider_name,
    d.name AS driver_name,
    t.pickup_location,
    t.drop_location,
    t.distance_km
FROM trips t
JOIN riders r ON t.rider_id = r.rider_id
JOIN drivers d ON t.driver_id = d.driver_id;

SELECT * FROM ride_details;


-- 5. TRIGGER
DELIMITER //

CREATE TRIGGER calculate_total_fare
BEFORE INSERT ON fares
FOR EACH ROW
BEGIN
    SET NEW.total_fare = NEW.base_fare + NEW.distance_charge;
END //

DELIMITER ;

SHOW TRIGGERS;
